## 3️⃣ measures.yaml 模板

```yaml
# 原子度量定义
# Measure 是对底层字段的基础聚合，是 Metric 的构建块

measures:
  # Measure 唯一标识
  net_revenue:
    # 显示名称
    name: 净销售收入
    
    # SQL 表达式
    expr: SUM(order_items.net_amount)
    
    # 所属表
    table: order_items
    
    # 是否可加（能否简单相加）
    additive: true  # true | false | semi_additive
    
    # 格式化
    format: currency
    currency: CNY
    precision: 2
    
    # NULL 处理
    null_handling: ignore  # ignore | zero | error
    
    # 聚合函数
    aggregation: sum  # sum | avg | count | count_distinct | min | max
    
    # 可聚合的维度（白名单）
    valid_dimensions:
      - region
      - product_category
      - customer
      - time
    
    # 时间可加性
    time_additive: true  # 可以跨时间相加
    
    # 描述
    description: 已支付订单的净销售金额（不含税、不含运费）
    
    # 元数据
    owner: finance_team
    certification: certified
    approved_by: CFO
    approved_at: 2026-08-15
    status: certified
    version: "1.2"
    created_at: 2026-01-01
    updated_at: 2026-08-15
  
  order_count:
    name: 有效订单数
    expr: COUNT(DISTINCT orders.id)
    table: orders
    additive: false  # 不能简单跨时间相加
    format: number
    precision: 0
    aggregation: count_distinct
    valid_dimensions:
      - region
      - product_category
      - customer
      - time
    time_additive: false  # 客户可能在多个时间段都有订单
    description: 已支付或已完成的唯一订单数
    owner: sales_analytics_team
    certification: certified
    status: certified
    version: "1.0"
  
  customer_count:
    name: 去重客户数
    expr: COUNT(DISTINCT customers.id)
    table: customers
    additive: false  # 不能跨时间相加
    format: number
    aggregation: count_distinct
    valid_dimensions:
      - region
      - customer_tier
      - time
    time_additive: false
    description: 唯一客户数（去重后）
    owner: growth_team
    certification: approved
    status: certified
    version: "1.0"
  
  gross_profit:
    name: 毛利润
    expr: SUM(order_items.net_amount - order_items.cost)
    table: order_items
    additive: true
    format: currency
    currency: CNY
    aggregation: sum
    valid_dimensions:
      - region
      - product_category
      - time
    time_additive: true
    description: 净销售额减去成本
    owner: finance_team
    certification: certified
    status: certified
    version: "1.0"
  
  units_sold:
    name: 销售件数
    expr: SUM(order_items.quantity)
    table: order_items
    additive: true
    format: number
    aggregation: sum
    valid_dimensions:
      - region
      - product
      - product_category
      - time
    time_additive: true
    description: 售出商品的总件数
    owner: sales_analytics_team
    certification: approved
    status: certified
    version: "1.0"
```

```yaml
      order_id:
        type: INTEGER
        description: 订单ID

        type: INTEGER

        type: INTEGER
```
