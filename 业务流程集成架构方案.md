# RDS Agent 业务流程集成架构方案

## 1. 需求分析

### 1.1 完整业务流程

```
┌─────────────┐    ┌──────────────┐    ┌──────────────┐    ┌──────────────┐
│ 数据查询     │───>│ 业务分析      │───>│ 业务操作      │───>│ 执行反馈      │
│ (RDS Agent) │    │ (Analysis    │    │ (Business    │    │ (API Calls)  │
│             │    │  Skills)     │    │  Skills)     │    │              │
└─────────────┘    └──────────────┘    └──────────────┘    └──────────────┘
      │                    │                    │                    │
      v                    v                    v                    v
  查询指标数据          分析数据趋势          决策业务动作          调用外部API
  生成SQL报表          异常检测诊断          自动化执行            更新业务系统
```

### 1.2 典型场景示例

#### 场景 1：销售异常检测与自动跟进
```
用户问题: "检查最近一周销售额是否异常，如有问题自动通知销售团队"

流程:
1. 数据查询: RDS Agent 查询最近一周各地区销售数据

3. 业务决策: 自动生成跟进任务
4. 业务操作: 调用 CRM API 创建销售跟进任务，发送钉钉/企微通知
```

#### 场景 2：库存预警与自动补货
```

流程:
1. 数据查询: 查询当前库存和销售速率
2. 业务分析: 计算预计断货时间，识别需要补货的 SKU
3. 业务决策: 根据补货策略计算补货量
4. 业务操作: 调用 ERP API 创建采购订单
```

#### 场景 3：用户流失预警与挽回
```
用户问题: "识别高价值流失风险用户并自动触发挽回活动"

流程:
1. 数据查询: 查询用户活跃度、消费金额、最后购买时间
2. 业务分析: RFM 分析，识别流失风险用户
3. 业务决策: 根据用户价值选择挽回策略（优惠券/专属客服）
```


### 2.2 分层职责

#### Layer 1: Orchestrator Layer（编排层）
- **职责：** 流程控制、状态管理、条件分支、错误处理
- **技术：** LangGraph / Temporal / Airflow
- **核心能力：**
  - 多步骤工作流编排
  - 条件判断和循环
  - 人工审批卡点
  - 失败重试和回滚

#### Layer 2: Skill Registry Layer（技能注册层）
- **职责：** 技能发现、注册、调用、版本管理
- **技术：** Plugin Framework / Dynamic Loading
- **核心能力：**
  - 统一 Skill 接口
  - 动态加载和热更新
  - 依赖管理
  - 权限控制

#### Layer 3: Execution Layer（执行层）
- **职责：** 具体任务执行、结果返回
- **技术：** RDS Agent / Python Functions / API Clients
- **核心能力：**
  - 数据查询
  - 分析计算
  - API 调用
  - 结果格式化

## 3. 核心组件设计

### 3.1 Skill 接口标准

#### 3.1.1 统一 Skill 接口

```python
from abc import ABC, abstractmethod
from typing import Dict, Any, List, Optional
from dataclasses import dataclass
from enum import Enum

class SkillType(Enum):
    """技能类型"""
    DATA_QUERY = "data_query"          # 数据查询
    ANALYSIS = "analysis"              # 业务分析
    DECISION = "decision"              # 决策判断
    OPERATION = "operation"            # 业务操作
    NOTIFICATION = "notification"      # 通知告警
    INTEGRATION = "integration"        # 外部集成

@dataclass
class SkillMetadata:
    """技能元数据"""
    name: str                          # 技能名称
    display_name: str                  # 显示名称
    description: str                   # 描述
    skill_type: SkillType             # 技能类型
    version: str                       # 版本号
    author: str                        # 作者
    dependencies: List[str] = None     # 依赖的其他 Skill
    required_permissions: List[str] = None  # 所需权限
    input_schema: Dict = None          # 输入 Schema
    output_schema: Dict = None         # 输出 Schema
    tags: List[str] = None             # 标签

@dataclass
class SkillContext:
    """技能执行上下文"""
    user_id: str                       # 用户 ID
    session_id: str                    # 会话 ID
    workflow_id: str                   # 工作流 ID
    previous_results: Dict[str, Any]   # 前序步骤结果
    global_config: Dict[str, Any]      # 全局配置
    
@dataclass
class SkillResult:
    """技能执行结果"""
    success: bool                      # 是否成功
    data: Any                          # 返回数据
    metadata: Dict[str, Any] = None    # 元数据
    error: Optional[str] = None        # 错误信息
    warnings: List[str] = None         # 警告信息
    next_actions: List[str] = None     # 建议的后续动作

class Skill(ABC):
    """技能基类"""
    
    def __init__(self):
        self._metadata: Optional[SkillMetadata] = None
    
    @property
    def metadata(self) -> SkillMetadata:
        """返回技能元数据"""
        if self._metadata is None:
            self._metadata = self._register_metadata()
        return self._metadata
    
    @abstractmethod
    def _register_metadata(self) -> SkillMetadata:
        """注册技能元数据（子类实现）"""
        pass
    
    @abstractmethod
    async def execute(
        self, 
        inputs: Dict[str, Any],
        context: SkillContext
    ) -> SkillResult:
        """执行技能（子类实现）"""
        pass
    
    async def validate_inputs(self, inputs: Dict[str, Any]) -> bool:
        """验证输入参数"""
        # 可选：子类可覆盖
        return True
    
    async def pre_execute(self, context: SkillContext) -> None:
        """执行前钩子（可选）"""
        pass
    
    async def post_execute(self, result: SkillResult, context: SkillContext) -> None:
        """执行后钩子（可选）"""
        pass
```

### 3.2 Skill Registry（技能注册中心）

```python
from typing import Dict, List, Type
import importlib
import inspect

class SkillRegistry:
    """技能注册中心"""
    
    def __init__(self):
        self._skills: Dict[str, Type[Skill]] = {}
        self._instances: Dict[str, Skill] = {}
    
    def register(self, skill_class: Type[Skill]) -> None:
        """注册技能"""
        skill_instance = skill_class()
        metadata = skill_instance.metadata
        
        # 验证依赖
        self._validate_dependencies(metadata)
        
        # 注册
        self._skills[metadata.name] = skill_class
        self._instances[metadata.name] = skill_instance
        
        print(f"✓ Skill registered: {metadata.name} (v{metadata.version})")
    
    def get_skill(self, skill_name: str) -> Optional[Skill]:
        """获取技能实例"""
        return self._instances.get(skill_name)
    
    def list_skills(
        self, 
        skill_type: Optional[SkillType] = None,
        tags: Optional[List[str]] = None
    ) -> List[SkillMetadata]:
        """列出技能"""
        skills = []
        for skill in self._instances.values():
            metadata = skill.metadata
            
            # 过滤类型
            if skill_type and metadata.skill_type != skill_type:
                continue
            
            # 过滤标签
            if tags and not any(tag in (metadata.tags or []) for tag in tags):
                continue
            
            skills.append(metadata)
        
        return skills
    
    def discover_and_register(self, package_path: str) -> None:
        """自动发现并注册技能"""
        module = importlib.import_module(package_path)
        
        for name, obj in inspect.getmembers(module):
            if (inspect.isclass(obj) and 
                issubclass(obj, Skill) and 
                obj is not Skill):
                self.register(obj)
    
    def _validate_dependencies(self, metadata: SkillMetadata) -> None:
        """验证依赖"""
        if not metadata.dependencies:
            return
        
        for dep in metadata.dependencies:
            if dep not in self._skills:
                raise ValueError(
                    f"Skill '{metadata.name}' depends on '{dep}', "
                    f"but '{dep}' is not registered"
                )

# 全局注册中心
skill_registry = SkillRegistry()
```

### 3.3 Workflow Orchestrator（工作流编排器）

```python
from typing import Dict, List, Any, Optional, Callable
from dataclasses import dataclass, field
from enum import Enum
import asyncio

class NodeType(Enum):
    """节点类型"""
    SKILL = "skill"              # 技能节点
    CONDITION = "condition"      # 条件节点
    PARALLEL = "parallel"        # 并行节点
    HUMAN_APPROVAL = "approval"  # 人工审批节点

@dataclass
class WorkflowNode:
    """工作流节点"""
    id: str
    name: str
    node_type: NodeType
    skill_name: Optional[str] = None           # Skill 节点使用
    condition_fn: Optional[Callable] = None    # Condition 节点使用
    inputs_mapping: Dict[str, str] = field(default_factory=dict)
    outputs_mapping: Dict[str, str] = field(default_factory=dict)
    retry_count: int = 3
    timeout_seconds: int = 300

@dataclass
class WorkflowEdge:
    """工作流边"""
    from_node: str
    to_node: str
    condition: Optional[Callable] = None       # 条件边

@dataclass
class Workflow:
    """工作流定义"""
    id: str
    name: str
    description: str
    nodes: List[WorkflowNode]
    edges: List[WorkflowEdge]
    entry_node: str
    global_config: Dict[str, Any] = field(default_factory=dict)

class WorkflowExecutor:
    """工作流执行器"""
    
    def __init__(self, skill_registry: SkillRegistry):
        self.skill_registry = skill_registry
        self.state: Dict[str, Any] = {}
    
    async def execute_workflow(
        self,
        workflow: Workflow,
        initial_inputs: Dict[str, Any],
        context: SkillContext
    ) -> Dict[str, Any]:
        """执行工作流"""
        self.state = {"initial_inputs": initial_inputs}
        
        # 从入口节点开始
        current_node_id = workflow.entry_node
        
        while current_node_id:
            node = self._get_node(workflow, current_node_id)
            
            # 执行节点
            result = await self._execute_node(node, workflow, context)
            
            # 保存结果
            self.state[node.id] = result
            
            # 确定下一个节点
            current_node_id = self._get_next_node(workflow, node.id, result)
        
        return self.state
    
    async def _execute_node(
        self,
        node: WorkflowNode,
        workflow: Workflow,
        context: SkillContext
    ) -> SkillResult:
        """执行节点"""
        if node.node_type == NodeType.SKILL:
            return await self._execute_skill_node(node, context)
        
        elif node.node_type == NodeType.CONDITION:
            return await self._execute_condition_node(node)
        
        elif node.node_type == NodeType.PARALLEL:
            return await self._execute_parallel_node(node, workflow, context)
        
        elif node.node_type == NodeType.HUMAN_APPROVAL:
            return await self._execute_approval_node(node)
        
        else:
            raise ValueError(f"Unknown node type: {node.node_type}")
    
    async def _execute_skill_node(
        self,
        node: WorkflowNode,
        context: SkillContext
    ) -> SkillResult:
        """执行技能节点"""
        skill = self.skill_registry.get_skill(node.skill_name)
        if not skill:
            raise ValueError(f"Skill not found: {node.skill_name}")
        
        # 映射输入
        inputs = self._map_inputs(node.inputs_mapping)
        
        # 更新上下文
        context.previous_results = self.state
        
        # 执行（带重试）
        for attempt in range(node.retry_count):
            try:
                result = await asyncio.wait_for(
                    skill.execute(inputs, context),
                    timeout=node.timeout_seconds
                )
                
                if result.success:
                    return result
                
                print(f"Skill execution failed (attempt {attempt + 1}): {result.error}")
            
            except asyncio.TimeoutError:
                print(f"Skill execution timeout (attempt {attempt + 1})")
            
            except Exception as e:
                print(f"Skill execution error (attempt {attempt + 1}): {e}")
        
        # 所有重试都失败
        return SkillResult(
            success=False,
            data=None,
            error=f"Skill '{node.skill_name}' failed after {node.retry_count} attempts"
        )
    
    async def _execute_condition_node(self, node: WorkflowNode) -> SkillResult:
        """执行条件节点"""
        result = node.condition_fn(self.state)
        return SkillResult(success=True, data={"condition_result": result})
    
    async def _execute_parallel_node(
        self,
        node: WorkflowNode,
        workflow: Workflow,
        context: SkillContext
    ) -> SkillResult:
        """执行并行节点"""
        # 获取所有并行子节点
        parallel_nodes = self._get_parallel_children(workflow, node.id)
        
        # 并行执行
        tasks = [
            self._execute_node(child_node, workflow, context)
            for child_node in parallel_nodes
        ]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        return SkillResult(success=True, data={"parallel_results": results})
    
    async def _execute_approval_node(self, node: WorkflowNode) -> SkillResult:
        """执行人工审批节点"""
        # 这里应该触发审批流程，等待人工决策
        # 简化实现：直接返回
        print(f"⏸️  Waiting for human approval for node: {node.name}")
        # TODO: 实现审批逻辑
        return SkillResult(success=True, data={"approved": True})
    
    def _map_inputs(self, mapping: Dict[str, str]) -> Dict[str, Any]:
        """映射输入参数"""
        inputs = {}
        for key, source_path in mapping.items():
            # 支持 JSONPath 风格：previous_results.node1.data.revenue
            value = self._get_nested_value(self.state, source_path)
            inputs[key] = value
        return inputs
    
    def _get_nested_value(self, data: Dict, path: str) -> Any:
        """获取嵌套值"""
        keys = path.split('.')
        current = data
        for key in keys:
            if isinstance(current, dict):
                current = current.get(key)
            else:
                return None
        return current
    
    def _get_node(self, workflow: Workflow, node_id: str) -> WorkflowNode:
        """获取节点"""
        for node in workflow.nodes:
            if node.id == node_id:
                return node
        raise ValueError(f"Node not found: {node_id}")
    
    def _get_next_node(
        self,
        workflow: Workflow,
        current_node_id: str,
        result: SkillResult
    ) -> Optional[str]:
        """确定下一个节点"""
        for edge in workflow.edges:
            if edge.from_node == current_node_id:
                # 检查条件
                if edge.condition is None or edge.condition(result):
                    return edge.to_node
        return None
    
    def _get_parallel_children(
        self,
        workflow: Workflow,
        parent_node_id: str
    ) -> List[WorkflowNode]:
        """获取并行子节点"""
        children = []
        for edge in workflow.edges:
            if edge.from_node == parent_node_id:
                child = self._get_node(workflow, edge.to_node)
                children.append(child)
        return children
```

### 4.2 异常检测 Skill

```python
class AnomalyDetectionSkill(Skill):
    """异常检测技能"""
    
    def _register_metadata(self) -> SkillMetadata:
        return SkillMetadata(
            name="anomaly_detection",
            display_name="异常检测",
            description="检测时间序列数据中的异常点",
            skill_type=SkillType.ANALYSIS,
            version="1.0.0",
            author="System",
            dependencies=["rds_query"],
            input_schema={
                "data": {"type": "array", "required": True},
                "metric_column": {"type": "string", "required": True},
                "threshold_std": {"type": "number", "default": 2.0}
            },
            output_schema={
                "has_anomaly": {"type": "boolean"},
                "anomalies": {"type": "array"},
                "analysis": {"type": "string"}
            },
            tags=["analysis", "anomaly", "detection"]
        )
    
    async def execute(
        self,
        inputs: Dict[str, Any],
        context: SkillContext
    ) -> SkillResult:
        """执行异常检测"""
        import numpy as np
        
        data = inputs["data"]
        metric_column = inputs["metric_column"]
        threshold_std = inputs.get("threshold_std", 2.0)
        
        # 提取指标值
        values = [row[metric_column] for row in data if metric_column in row]
        
        if len(values) < 3:
            return SkillResult(
                success=True,
                data={
                    "has_anomaly": False,
                    "anomalies": [],
                    "analysis": "数据点不足，无法进行异常检测"
                }
            )
        
        # 计算均值和标准差
        mean_val = np.mean(values)
        std_val = np.std(values)
        
        # 检测异常
        anomalies = []
        for i, (row, value) in enumerate(zip(data, values)):
            z_score = abs(value - mean_val) / std_val if std_val > 0 else 0
            
            if z_score > threshold_std:
                anomalies.append({
                    "index": i,
                    "data": row,
                    "value": value,
                    "z_score": float(z_score),
                    "deviation_pct": float((value - mean_val) / mean_val * 100)
                })
        
        # 生成分析报告
        if anomalies:
            analysis = f"检测到 {len(anomalies)} 个异常点。"
            analysis += f"均值: {mean_val:.2f}, 标准差: {std_val:.2f}"
        else:
            analysis = f"未检测到异常。均值: {mean_val:.2f}, 标准差: {std_val:.2f}"
        
        return SkillResult(
            success=True,
            data={
                "has_anomaly": len(anomalies) > 0,
                "anomalies": anomalies,
                "analysis": analysis,
                "statistics": {
                    "mean": float(mean_val),
                    "std": float(std_val),
                    "count": len(values)
                }
            },
            next_actions=["notify_team"] if anomalies else []
        )

# 注册
```

```python
class APICallSkill(Skill):
    """外部 API 调用技能"""
    
    def _register_metadata(self) -> SkillMetadata:
        return SkillMetadata(
            name="api_call",
            display_name="API 调用",
            description="调用外部业务系统 API",
            skill_type=SkillType.INTEGRATION,
            version="1.0.0",
            author="System",
            required_permissions=["api:call"],
            input_schema={
                "api_name": {"type": "string", "required": True},
                "method": {"type": "string", "enum": ["GET", "POST", "PUT", "DELETE"]},
                "params": {"type": "object"},
                "body": {"type": "object"}
            },
            tags=["integration", "api", "external"]
        )
    
    async def execute(
        self,
        inputs: Dict[str, Any],
        context: SkillContext
    ) -> SkillResult:
        """调用 API"""
        import aiohttp
        
        api_name = inputs["api_name"]
        method = inputs.get("method", "POST")
        params = inputs.get("params", {})
        body = inputs.get("body", {})
        
        # 从配置中获取 API 定义
        api_registry = context.global_config.get("api_registry", {})
        api_config = api_registry.get(api_name)
        
        if not api_config:
            return SkillResult(
                success=False,
                data=None,
                error=f"API '{api_name}' not found in registry"
            )
        
        # 构造请求
        url = api_config["base_url"] + api_config["path"]
        headers = api_config.get("headers", {})
        
        # 添加认证
        if "auth" in api_config:
            auth_type = api_config["auth"]["type"]
            if auth_type == "bearer":
                token = api_config["auth"]["token"]
                headers["Authorization"] = f"Bearer {token}"
        
        try:
            async with aiohttp.ClientSession() as session:
                async with session.request(
                    method,
                    url,
                    params=params,
                    json=body if body else None,
                    headers=headers,
                    timeout=aiohttp.ClientTimeout(total=30)
                ) as resp:
                    response_data = await resp.json()
                    
                    if resp.status >= 200 and resp.status < 300:
                        return SkillResult(
                            success=True,
                            data=response_data,
                            metadata={
                                "api_name": api_name,
                                "status_code": resp.status,
                                "method": method
                            }
                        )
                    else:
                        return SkillResult(
                            success=False,
                            data=response_data,
                            error=f"API returned {resp.status}: {response_data}"
                        )
        
        except Exception as e:
            return SkillResult(
                success=False,
                data=None,
                error=f"API call failed: {str(e)}"
            )
```

```python
# workflow_mcp_server.py
from mcp.server.fastmcp import FastMCP
from mcp.server.fastmcp.exceptions import ToolError
import json

mcp = FastMCP("business-workflow")

@mcp.tool()
async def execute_workflow(
    workflow_name: str,
    inputs: str  # JSON string
) -> str:
    """执行业务工作流
    
    Args:
        workflow_name: 工作流名称，如 'sales_anomaly_detection'
        inputs: JSON 格式的输入参数
    
    Returns:
        工作流执行结果
    """
    try:
        # 解析输入
        initial_inputs = json.loads(inputs)
        
        # 获取工作流
        workflow = get_workflow_by_name(workflow_name)
        if not workflow:
            raise ToolError(f"Workflow '{workflow_name}' not found")
        
        # 执行
        executor = WorkflowExecutor(skill_registry)
        context = SkillContext(
            user_id="mcp_user",
            session_id=f"mcp_{workflow_name}",
            workflow_id=workflow.id,
            previous_results={},
            global_config=workflow.global_config
        )
        
        result = await executor.execute_workflow(workflow, initial_inputs, context)
        
        # 格式化返回
        return json.dumps({
            "success": True,
            "workflow_id": workflow.id,
            "result": result
        }, ensure_ascii=False, indent=2)
    
    except Exception as e:
        raise ToolError(f"Workflow execution failed: {str(e)}")

@mcp.tool()
async def list_workflows() -> str:
    """列出所有可用的业务工作流"""
    workflows = get_all_workflows()
    return json.dumps([
        {
            "id": wf.id,
            "name": wf.name,
            "description": wf.description
        }
        for wf in workflows
    ], ensure_ascii=False, indent=2)

    """列出所有可用的业务技能
    
    Args:
        skill_type: 技能类型筛选（可选），如 'analysis', 'operation'
    """
    skill_type_enum = SkillType(skill_type) if skill_type else None

    return json.dumps([
        {
            "name": skill.name,
            "display_name": skill.display_name,
            "description": skill.description,
            "type": skill.skill_type.value,
            "version": skill.version
        }
        for skill in skills
    ], ensure_ascii=False, indent=2)
```
